pybind11_add_module(mace_foundation_bindings MODULE bindings.cpp cpu.cpp)
set_target_properties(mace_foundation_bindings PROPERTIES OUTPUT_NAME _mace_foundation)
target_include_directories(mace_foundation_bindings PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../libsymmetrix/external/json/single_include ${CMAKE_CURRENT_SOURCE_DIR}/../../../../libsymmetrix/source)
target_compile_options(mace_foundation_bindings PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-march=native>)
set_source_files_properties(cpu.cpp PROPERTIES COMPILE_OPTIONS "$<$<AND:$<CONFIG:Release>,$<NOT:$<CXX_COMPILER_ID:NVHPC>>>:-ffast-math;-fno-finite-math-only>")
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(mace_foundation_bindings PRIVATE OpenMP::OpenMP_CXX)
endif()

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(mace_foundation_bindings PRIVATE cuda.cu)
  target_compile_definitions(mace_foundation_bindings PRIVATE MACE_FOUNDATION_CUDA)
  target_link_libraries(mace_foundation_bindings PRIVATE CUDA::cudart CUDA::cublas)
  set_source_files_properties(cuda.cu PROPERTIES COMPILE_OPTIONS "--expt-relaxed-constexpr;$<$<CONFIG:Release>:--use_fast_math>")
  set_target_properties(mace_foundation_bindings PROPERTIES CUDA_STANDARD 20 CUDA_ARCHITECTURES "${SYMMETRIX_CUDA_ARCHITECTURES}")
endif()

install(TARGETS mace_foundation_bindings DESTINATION symmetrix)
